/*  배치 로그                                                                */
/* ######################################################################### */
/* =================================== */
/*  배치로그 - 프로시저별 확인         */
/* =================================== */
SELECT * FROM RPR_PROG_EXEC_H
 WHERE 1 = 1
   AND SUB_ID IN ('RPR_DAILY_PROGRESS_SP', 'RPR_DAILY_PROGRESS_SP')
 ORDER BY CRT_TM DESC
;

/* =================================== */
/*  배치 실행시간 확인                 */
/* =================================== */
SELECT REV_NO,MAIN_ID,PLN_DT,SUB_STAT_CD
			, START_TM, END_TM
     , ROUND(TO_DATE(END_TM, 'YYYY-MM-DD HH24:MI:SS')
           - TO_DATE(START_TM, 'YYYY-MM-DD HH24:MI:SS'), 2) AS dd
     , ROUND((TO_DATE(END_TM, 'YYYY-MM-DD HH24:MI:SS')
            - TO_DATE(START_TM, 'YYYY-MM-DD HH24:MI:SS')) * 24, 2) AS hh
     , ROUND((TO_DATE(END_TM, 'YYYY-MM-DD HH24:MI:SS')
            - TO_DATE(START_TM, 'YYYY-MM-DD HH24:MI:SS')) * 24 * 60, 2) AS mm
     , ROUND((TO_DATE(END_TM, 'YYYY-MM-DD HH24:MI:SS')
            - TO_DATE(START_TM, 'YYYY-MM-DD HH24:MI:SS')) * 24 * 60 * 60, 2) AS ss
  FROM RPR_PROG_EXEC_H
 WHERE SUB_ID IN ('RPR_DAILY_PROGRESS_SP', 'RPR_DAILY_PROGRESS_SP')
 ORDER BY CRT_TM DESC
;

/* =================================== */
