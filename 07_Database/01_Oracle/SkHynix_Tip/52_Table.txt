       VO.XIDUSN, VO.LOCKED_MODE
  FROM V$LOCKED_OBJECT VO , DBA_OBJECTS DO
 WHERE VO.OBJECT_ID = DO.OBJECT_ID
;

SELECT COUNT(*)
  FROM V$LOCKED_OBJECT VO , DBA_OBJECTS DO
 WHERE VO.OBJECT_ID = DO.OBJECT_ID
;

SELECT A.SID, A.SERIAL#
  FROM V$SESSION A, V$LOCK B, DBA_OBJECTS C
 WHERE A.SID=B.SID
   AND B.ID1=C.OBJECT_ID
   AND B.TYPE='TM'
